# Skyflow Python SDK — v2 (PDB) and v3 (FlowDB) Installation

## v2 — PDB SDK

### Install

```bash
pip install "skyflow-python==2.0.0"
```

### Initialize client

```python
from skyflow import Skyflow, LogLevel

client = (
    Skyflow.builder()
    .add_vault_config({
           'vault_id': '<VAULT_ID>',
           'cluster_id': '<CLUSTER_ID>',
           'env': Env.PROD,
           'credentials': credentials
    })
    .set_log_level(LogLevel.INFO)
    .build()
)
```

### Insert

```python
from skyflow import InsertRequest

response = client.vault().insert(
    InsertRequest(
        table="cards",
        values=[{"card_number": "4111111111111111", "cardholder_name": "John Doe"}],
        return_tokens=True,
    )
)
print(response)
# {"records": [{"skyflow_id": "...", "table": "cards"}], "errors": []}
```

# Skyflow Node.js SDK — v2 (PDB) and v3 (FlowDB) Installation

## v2 — PDB SDK

### Install

```bash
npm install skyflow-node
```

### Initialize client

```javascript
import { Skyflow, SkyflowConfig, VaultConfig, Env, LogLevel } from 'skyflow-node';

const skyflowCredentials = {
    apiKey: "<SKYFLOW_API_KEY>"
};

const vaultConfig: VaultConfig = {
    vaultId: '<VAULT_ID>',
    clusterId: '<CLUSTER_ID>',
    env: Env.PROD,
};

const skyflowConfig: SkyflowConfig = {
    vaultConfigs: [vaultConfig],
    skyflowCredentials: skyflowCredentials,
    logLevel: LogLevel.ERROR
};

const skyflowClient: Skyflow = new Skyflow(skyflowConfig);
```

### Insert

```javascript
import { InsertRequest, InsertOptions } from "skyflow-node";

const insertData = [
  { card_number: "4111111111111112", cardholder_name: "John Doe" },
];
const insertReq = new InsertRequest("table1", insertData);

const insertOptions = new InsertOptions();
insertOptions.setReturnTokens(true);

const insertResponse = await skyflowClient
  .vault(vaultId)
  .insert(insertReq, insertOptions);

console.log("Insert response:", insertResponse);
// { insertedFields: [{ skyflowId: '...', card_number: '...', cardholder_name: '...' }], errors: null }
```
